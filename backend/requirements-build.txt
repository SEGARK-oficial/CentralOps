# Build-time only — used by the Cython compile stage to produce .so
# extension modules from .py sources before they enter the final image.
# Not installed into the runtime image (the compiled .so files do not
# depend on Cython at runtime).
# TETO FECHADO em 3.3 — o range antigo (`<4`) era frouxo e o Cython 3.3.0
# quebra a compilação deste projeto: `CompileError` SEM mensagem (o Cython
# levanta `CompileError(None, pyx_file)` e o traceback termina em
# `TypeError: NoneType object is not iterable`), reportada contra
# `app/collectors/registry.py` — mas o nome é enganoso, porque `cythonize`
# compila num pool de processos e atribui o erro ao item errado.
#
# Reproduzido: `3d2e4a1a`, que passou no CI em 19/08, falha hoje sem NENHUMA
# mudança de código — é a dependência que se moveu, não o repositório. Com
# este teto, os mesmos 233 módulos compilam.
#
# Subir o teto exige provar o build do estágio `backend-compile` numa máquina
# SEM cache de layer; o CI do Core mascarou a quebra por ter cache, e quem
# pagou foi o CI do EE, que builda limpo.
cython>=3.0,<3.3
